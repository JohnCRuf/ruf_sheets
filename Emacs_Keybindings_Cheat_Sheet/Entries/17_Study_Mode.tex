\section{Study Mode \quad \key{SPC n S}}

\key{SPC n S} $\to$ pick a subject $\to$ the world tree of curriculum progress
$\to$ enter a world. You get a submission buffer with the tutor's transcript
beside it. The tutor grades against a stored solution without showing it to
you.

\subsection{In the submission buffer}
\begin{keys}
\krow{C-c C-c}{submit the attempt for grading}
\krow{C-c C-h}{the hint ladder}
\krow{C-c C-n}{next problem \quad \key{C-c C-e} an easier one}
\krow{C-c C-a}{ask the tutor a free-form question}
\krow{C-c C-m}{materials for this problem}
\krow{C-c C-s}{the field's cheat sheet}
\krow{C-c ?}{how to type math here}
\krow{C-c C-t}{switch between flashcards and long form}
\krow{C-c C-d}{dictate the argument instead of typing it}
\krow{C-c C-g}{record the result by hand}
\krow{C-c C-q}{restart the tutor if it wedges}
\end{keys}

\subsection{The hint ladder \quad \key{C-c C-h}}
Escalate only after a real attempt.
\begin{keys}[1.6cm]
\krow{1}{redirect: a concept or a reference to look at}
\krow{2}{structure: the skeleton of the argument}
\krow{3}{one step worked, then handed back to you}
\end{keys}
Without spending a hint:
\begin{keys}[1.6cm]
\krow{c}{am I on track? (no grading)}
\krow{a}{ask a question}
\krow{m}{materials \quad \key{e} deal an easier problem}
\end{keys}

\subsection{The world tree}
\begin{keys}[1.6cm]
\krow{RET}{enter the world at point}
\krow{m}{its materials \quad \key{c} the cheat sheet}
\krow{l}{lectures \quad \key{u} unpack more problems}
\krow{x}{sit an exam \quad \key{W} review your weak spots}
\krow{g}{refresh \quad \key{q} quit}
\end{keys}
In the field list: \key{RET} enter, \key{N} add a field, \key{l} lectures,
\key{g} refresh.

\subsection{Handwriting on the tablet}
\begin{keys}
\krow{C-c C-r}{send this problem to the reMarkable}
\krow{C-c C-i}{import the handwritten attempt back}
\end{keys}

\subsection{Reviewing what the tutor wrote}
\begin{keys}
\krow{C-c C-c}{accept the suggested edit}
\krow{C-c C-k}{discard it}
\end{keys}
