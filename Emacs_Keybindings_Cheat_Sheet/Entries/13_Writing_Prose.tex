\section{Writing Prose}

\subsection{Get out of the way}
\begin{keys}
\krow{SPC t z}{olivetti: center the text column}
\krow{SPC m R}{read mode, in org and markdown}
\krow{SPC t w}{soft line wrapping}
\krow{SPC t l}{line numbers off (or on)}
\krow{SPC t r}{read-only}
\end{keys}

\subsection{Spelling and shape}
\begin{keys}
\krow{z =}{correct the word at point}
\krow{Z =}{change the spellcheck language}
\krow{g q i p}{re-wrap this paragraph}
\krow{g w}{re-wrap without moving point}
\end{keys}

\subsection{Markdown}
\begin{keys}
\krow{SPC m P}{render this file to PDF}
\krow{SPC m R}{read mode}
\krow{SPC m e f}{Substack preflight check}
\krow{SPC m e h}{draft history for this file}
\end{keys}

\subsection{Publishing to Substack \quad \key{SPC m e}}
\begin{keys}
\krow{SPC m e s}{push as a draft}
\krow{SPC m e p}{publish}
\krow{SPC m e n}{post as a note}
\krow{SPC m e S}{export to the clipboard instead}
\krow{SPC n w n}{start a new post in the vault}
\krow{SPC o s}{the Substack action menu}
\krow{SPC o S}{sync the list of accounts you follow}
\end{keys}
Run the preflight (\key{SPC m e f}) before pushing: it is what catches the
markup Substack silently mangles.

\subsection{LaTeX documents}
\begin{keys}
\krow{SPC m s}{side-by-side source and PDF}
\krow{SPC m c}{compile \quad \key{SPC m a} run the whole chain}
\krow{SPC m v}{jump to this point in the PDF}
\krow{SPC m m}{pick an AUCTeX command by hand}
\krow{SPC m p \ SPC m P}{preview / unpreview at point}
\krow{C-c C-c}{run the next sensible command}
\end{keys}
